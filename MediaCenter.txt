import React, { useState, useEffect } from 'react'
import { posterUrl } from '../lib/api.js'
import styles from './MediaCenter.module.css'

export default function ActorDetails({ actor, activeMedia, onBack }) {
  const [details, setDetails] = useState(null)
  const [credits, setCredits] = useState([])

  useEffect(() => {
    if (!actor) return
    
    // Получаем биографию актера
    fetch(`https://themoviedb.org{actor.id}?api_key=c11ba9b88c992211a2266d8e1283fe18`)
      .then(res => res.ok ? res.json() : null)
      .then(data => setDetails(data))
      .catch(console.error)

    // Получаем известные фильмы актера
    fetch(`https://themoviedb.org{actor.id}/combined_credits?api_key=c11ba9b88c992211a2266d8e1283fe18`)
      .then(res => res.ok ? res.json() : null)
      .then(data => {
        const topCredits = data?.cast?.sort((a, b) => b.popularity - a.popularity).slice(0, 5) || []
        setCredits(topCredits)
      })
      .catch(console.error)
  }, [actor])

  return (
    <div className={styles.assetPage}>
      <button className={styles.backBtn} onClick={onBack}>← Back to Media</button>
      
      {activeMedia?.backdrop_path && (
        <img src={posterUrl(activeMedia.backdrop_path, true)} alt="" className={styles.blurredBackground} referrerPolicy="no-referrer" />
      )}

      <div className={styles.articleContainer}>
        <div className={styles.leftColumn}>
          <h1 className={styles.assetTitleFull}>{actor.name}</h1>
          <div className={styles.synopsisBox}>
            <h3 className={styles.boxLabel}>Biography</h3>
            <p className={styles.synopsisText} style={{ whiteSpace: 'pre-line' }}>
              {details?.biography || "Biography is being updated."}
            </p>

            <div className={styles.metaRow} style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '15px', marginTop: '25px' }}>
              <div><strong>Birthday:</strong> {details?.birthday || 'N/A'}</div>
              <div><strong>Place of Birth:</strong> {details?.place_of_birth || 'N/A'}</div>
              <div><strong>Known For:</strong> {details?.known_for_department || 'Acting'}</div>
              <div><strong>Popularity Score:</strong> {details?.popularity?.toFixed(1) || 'N/A'}</div>
            </div>

            <div className={styles.castSection} style={{ marginTop: '30px' }}>
              <h3 className={styles.boxLabel}>Known For</h3>
              <div className={styles.castList}>
                {credits.map((media, idx) => (
                  <div key={idx} className={styles.mediaCard} style={{ flex: '0 0 110px', padding: '6px', background: 'rgba(255,255,255,0.02)', cursor: 'default' }}>
                    <img src={media.poster_path ? posterUrl(media.poster_path) : 'https://placehold.co'} alt="" style={{ width: '100%', borderRadius: '4px', aspectRatio: '2/3', objectFit: 'cover' }} />
                    <h4 style={{ fontSize: '11px', marginTop: '6px', whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis', color: '#fff' }}>{media.title || media.name}</h4>
                  </div>
                ))}
              </div>
            </div>
          </div>
        </div>

        <div className={styles.rightColumn}>
          <div className={styles.posterWrapper} style={{ maxWidth: '280px', borderRadius: '8px' }}>
            <img src={actor.profile_path ? posterUrl(actor.profile_path, true) : 'https://placehold.co'} alt={actor.name} className={styles.mainAssetImg} referrerPolicy="no-referrer" />
          </div>
        </div>
      </div>
    </div>
  )
}
